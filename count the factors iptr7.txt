import java.util.*;
public class Main{
    public static void main(String[]args){
        Scanner Scan = new Scanner(System.in);
        int num = Scan.nextInt();
        int Count = 0;
        for(int i=2;i<=num/2;i++){
            if(num%i==0){
                Count++;
            }
        }
        System.out.println(Count+2);
    }
}
