import java.util.*;
public class Main{
    public static void main(String[]args){
        Scanner Scan = new Scanner(System.in);
        int num = Scan.nextInt();
        int a = 0;
        int b = 1;
        System.out.println(a+""+b+"");
        for(int i=2;i<+num;i++){
            int c=a+b;
            System.out.println(c+"");
            a=b;
            b=c;
        }
    }
}
