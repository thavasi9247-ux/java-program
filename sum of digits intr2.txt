import java.util.*;
public class Main{
    public static void main(String[]args){
        Scanner Scan = new Scanner(System.in);
        int a = Scan.nextInt();
        int sum = 0;
        while(a>0){
            int r = a%10;
            sum = sum+r;
            a=a/10;
        }
        System.out.println(sum);
    }
}
