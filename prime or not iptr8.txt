import java.util.*;
public class Main{
    public static void main(String[]args){
        Scanner Scan = new Scanner(System.in);
        int num = Scan.nextInt();
        int Count = 0;
        for(int i=1;i<=num;i++){
            if(num%i==0){
                Count++;
            }
        }
        if(Count==2){
            System.out.println("prime");
        }
        else{
            System.out.println("not a prime");
        }
    }
}
