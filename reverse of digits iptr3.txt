import java.util.*;
public class Main{
    public static void main(String[]args){
        Scanner Scan = new Scanner(System.in);
        int a = Scan.nextInt();
        int rev = 0;
        while(a>0){
            int r = a%10;
            rev = rev*10+r;
            a=a/10;
        }
        System.out.println(rev);
    }
}
